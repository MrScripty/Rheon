//! Bounded pressure-solve/path counterexample using public fitted assembly.
//! This diagnostic never owns or publishes an advancing liquid state.
use rheon::*;

const N: usize = 16;
const V: usize = 22;
const P: usize = 16;
const T: usize = 24;
const S: usize = V + P;
const DT: f64 = 0.05;
const CAP: [[f64; 2]; 3] = [[0., 1.], [0.5, 1.25], [1., 1.]];

struct Assembly {
    r: [[[f64; V]; 3]; N],
    mass: [[f64; V]; V],
    strain: [[f64; V]; V],
    divergence: [[f64; V]; T],
    pressure: [[f64; P]; T],
    b: [[f64; V]; P],
}
struct SolveScratch {
    original: [[f64; S]; S],
    factors: [[f64; S]; S],
    rhs: [f64; S],
    eliminated: [f64; S],
    solution: [f64; S],
}
fn frame(cap: &[[f64; 2]; 3]) -> Result<FittedHeightWorkspace, FittedHeightError> {
    FittedHeightWorkspace::new(
        FittedHeightGeometry {
            cap,
            bottom_x: &[0., 0.5, 1.],
            extrusion_width: 1.,
            density: 3.,
            dynamic_viscosity: 0.05,
        },
        Default::default(),
        |_| false,
    )
}
fn assemble(w: &FittedHeightWorkspace) -> Assembly {
    assert_eq!(w.plan().periodic_nodes, N);
    assert_eq!(w.plan().triangles, T);
    assert_eq!(w.plan().pressure_modes, P);
    let mut a = Assembly {
        r: [[[0.; V]; 3]; N],
        mass: [[0.; V]; V],
        strain: [[0.; V]; V],
        divergence: [[0.; V]; T],
        pressure: [[0.; P]; T],
        b: [[0.; V]; P],
    };
    for (i, row) in a.r.iter_mut().enumerate() {
        for (d, component) in row.iter_mut().enumerate().take(2) {
            let e = w.velocity_embedding(i, d).unwrap();
            for k in 0..2 {
                if let Some(j) = e.columns[k] {
                    assert!(j < V);
                    component[j] += e.weights[k];
                }
            }
        }
        for j in 0..V {
            for k in 0..V {
                for component in row.iter().take(2) {
                    a.mass[j][k] += w.nodal_mass()[i] * component[j] * component[k];
                }
            }
        }
    }
    for term in w.pressure_basis() {
        a.pressure[term.triangle][term.mode] = term.value;
    }
    for (t, tri) in w.triangles().iter().enumerate() {
        let stiffness = w.triangle_stiffness(t).unwrap();
        let nodes = tri.nodes.map(|i| w.nodes()[i].periodic_index);
        for (i, &node) in nodes.iter().enumerate() {
            for j in 0..V {
                for d in 0..2 {
                    a.divergence[t][j] += tri.gradients[i][d] * a.r[node][d][j];
                }
            }
        }
        for j in 0..V {
            for k in 0..V {
                for l in 0..9 {
                    for h in 0..9 {
                        a.strain[j][k] += a.r[nodes[l / 3]][l % 3][j]
                            * stiffness[l][h]
                            * a.r[nodes[h / 3]][h % 3][k];
                    }
                }
            }
        }
        for p in 0..P {
            for j in 0..V {
                a.b[p][j] -= a.pressure[t][p] * tri.area * a.divergence[t][j];
            }
        }
    }
    a
}
fn solve(a: &Assembly, old: &[f64; V]) -> Result<([f64; S], f64), &'static str> {
    let mut s = SolveScratch {
        original: [[0.; S]; S],
        factors: [[0.; S]; S],
        rhs: [0.; S],
        eliminated: [0.; S],
        solution: [0.; S],
    };
    for i in 0..V {
        for (j, &value) in old.iter().enumerate() {
            s.original[i][j] = a.mass[i][j] + DT * a.strain[i][j];
            s.rhs[i] += a.mass[i][j] * value;
        }
        for p in 0..P {
            s.original[i][V + p] = DT * a.b[p][i];
            s.original[V + p][i] = a.b[p][i];
        }
    }
    s.factors = s.original;
    s.eliminated = s.rhs;
    // Fixed 38x38 host diagnostic, never a production pressure solver.
    for j in 0..S {
        let pivot = (j..S)
            .max_by(|&i, &k| s.factors[i][j].abs().total_cmp(&s.factors[k][j].abs()))
            .unwrap();
        s.factors.swap(j, pivot);
        s.eliminated.swap(j, pivot);
        if !s.factors[j][j].is_normal() {
            return Err("invalid pressure diagnostic pivot");
        }
        for i in j + 1..S {
            let factor = s.factors[i][j] / s.factors[j][j];
            for k in j..S {
                s.factors[i][k] -= factor * s.factors[j][k];
            }
            s.eliminated[i] -= factor * s.eliminated[j];
        }
    }
    for i in (0..S).rev() {
        let tail: f64 = (i + 1..S).map(|j| s.factors[i][j] * s.solution[j]).sum();
        s.solution[i] = (s.eliminated[i] - tail) / s.factors[i][i];
        if !s.solution[i].is_finite() || (s.solution[i] != 0. && !s.solution[i].is_normal()) {
            return Err("invalid pressure diagnostic solution");
        }
    }
    let residual = (0..S)
        .map(|i| {
            let row: f64 = (0..S).map(|j| s.original[i][j] * s.solution[j]).sum();
            (row - s.rhs[i]).abs()
        })
        .fold(0., f64::max);
    if residual > 1e-11 {
        return Err("pressure diagnostic true residual");
    }
    Ok((s.solution, residual))
}
fn energy(a: &Assembly, z: &[f64]) -> f64 {
    (0..V)
        .map(|i| (0..V).map(|j| z[i] * a.mass[i][j] * z[j]).sum::<f64>())
        .sum::<f64>()
        * 0.5
}
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut w = frame(&CAP)?;
    let a = assemble(&w);
    let mut old = [0.; V];
    for n in w.nodes() {
        let e = w.velocity_embedding(n.periodic_index, 0).unwrap();
        if e.weights[0] == 1. {
            old[e.columns[0].unwrap()] = n.position[1];
        }
    }
    let (solution, residual) = solve(&a, &old)?;
    let mut z = [0.; 34];
    z[..V].copy_from_slice(&solution[..V]);
    let mut u = [[0.; 3]; N];
    w.embed_velocity(&z, &mut u)?;
    let pressure: [f64; P] = solution[V..].try_into()?;
    let cap_velocity = [3, 4, 5].map(|i| {
        let v = u[w.nodes()[i].periodic_index];
        [v[0], v[1]]
    });
    let mut out = [FittedHeightNodeDiagnostic::default(); N];
    let report = w.inspect(
        FittedHeightInputs {
            velocity: &u,
            pressure_coefficients: &pressure,
        },
        &mut out,
        |_| false,
    )?;
    let delta: [f64; V] = std::array::from_fn(|i| solution[i] - old[i]);
    let e_old = energy(&a, &old);
    let e_new = energy(&a, &solution[..V]);
    let increment = energy(&a, &delta);
    let work = e_new - e_old + increment + DT * report.strain_power;
    if report.divergence_max > 1e-11 || work.abs() > 1e-11 || report.pressure_work.abs() > 1e-11 {
        return Err("pressure diagnostic divergence or full strain work".into());
    }
    println!(
        "{{\"scope\":\"native pressure force solve and rejected material path; no advancing state\",\"advancing_state_published\":false,\"dt\":{DT:?},\"frame_bytes\":{},\"fixed_assembly_bytes\":{},\"fixed_solver_bytes\":{},\"pressure_columns\":{:?},\"reduced_velocity\":{:?},\"pressure_coefficients\":{:?},\"cap_velocity\":{:?},\"momentum_residual\":{residual:?},\"energy_old\":{e_old:?},\"energy_new\":{e_new:?},\"increment_loss\":{increment:?},\"strain_power\":{:?},\"pressure_work\":{:?},\"work_error\":{work:?},\"samples\":[",
        w.allocated_bytes(),
        std::mem::size_of::<Assembly>(),
        std::mem::size_of::<SolveScratch>(),
        w.pressure_columns(),
        &solution[..V],
        pressure,
        cap_velocity,
        report.strain_power,
        report.pressure_work,
    );
    // Each path point is rebuilt independently, using a bounded Frame owner.
    // These are rejected candidate samples, never accepted simulation clocks.
    drop(w);
    for (case, fraction) in [0., 0.25, 0.5, 0.75, 1.].into_iter().enumerate() {
        let time = DT * fraction;
        let cap =
            std::array::from_fn(|i| std::array::from_fn(|d| CAP[i][d] + time * cap_velocity[i][d]));
        let mut sample = frame(&cap)?;
        sample.embed_velocity(&z, &mut u)?;
        // Pressure coefficients apply only on the original force mesh. The
        // later path samples use zero pressure for their continuity diagnostic.
        let p = if case == 0 { pressure } else { [0.; P] };
        let r = sample.inspect(
            FittedHeightInputs {
                velocity: &u,
                pressure_coefficients: &p,
            },
            &mut out,
            |_| false,
        )?;
        println!(
            "{}{{\"path_parameter\":{time:?},\"nodes\":{:?},\"triangles\":{:?},\"mass\":{:?},\"velocity\":{:?},\"divergence\":{:?},\"pressure_values\":{:?},\"mass_rates\":{:?},\"continuity_defect\":{:?},\"flux\":{:?},\"divergence_max\":{:?},\"continuity_defect_max\":{:?},\"total_mass\":{:?},\"geometric_identity_error\":{:?}}}",
            if case == 0 { "" } else { "," },
            sample
                .nodes()
                .iter()
                .map(|n| [n.position[0], n.position[1], n.periodic_index as f64])
                .collect::<Vec<_>>(),
            sample
                .triangles()
                .iter()
                .map(|t| t.nodes)
                .collect::<Vec<_>>(),
            sample.nodal_mass(),
            u,
            sample
                .triangle_scratch()
                .iter()
                .map(|s| s[0])
                .collect::<Vec<_>>(),
            sample
                .triangle_scratch()
                .iter()
                .map(|s| s[1])
                .collect::<Vec<_>>(),
            out.map(|o| o.mass_rate),
            out.map(|o| o.mass_rate + o.mass_flux_sum),
            sample
                .shared_flux_scratch()
                .iter()
                .map(|f| [f.nodes[0] as f64, f.nodes[1] as f64, f.flux])
                .collect::<Vec<_>>(),
            r.divergence_max,
            r.continuity_defect_max,
            r.total_mass,
            r.geometric_identity_error,
        );
    }
    println!("]}}");
    Ok(())
}
