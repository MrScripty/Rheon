"""Exact row identities and rational interval rank/geometry certificate.

The interval path uses the published binary coefficients as exact rationals.
This certifies their real polynomial path, not IEEE execution or any mesh family.
"""
from fractions import Fraction as Q
from functools import lru_cache
import inspect
import ast
import json
import sympy as sp
import model as m


@lru_cache(maxsize=1)
def row_identities():
    x0,x1,h0=sp.symbols('x0 x1 h0',real=True)
    cap=[(x0,h0),(x1,sp.Rational(9,4)-h0),(x0+1,h0)]
    source=inspect.getsource(m.geometry).replace('def geometry(', 'def formal_geometry(')
    # Formal rational expressions require nonzero denominators; physical path
    # positivity and those denominators are separately certified below.
    tree=ast.parse(source)
    function=tree.body[0]
    function.body=[n for n in function.body if not (isinstance(n,ast.Expr) and isinstance(n.value,ast.Call) and isinstance(n.value.func,ast.Name) and n.value.func.id=='require')]
    ns={'base':m.base,'Q':Q}
    exec(compile(tree,'<formal PS geometry>','exec'),ns)
    points,micro,ids,_,_=ns['formal_geometry'](cap,[(Q(0),Q(0))]*3)
    D=sp.zeros(24,22);areas=[]
    for t,tri in enumerate(micro):
        p=[points[i]for i in tri]
        twice=m.base.cross(m.base.sub(p[1],p[0]),m.base.sub(p[2],p[0])).v
        areas.append(twice/2)
        for i in range(3):
            j,k=(i+1)%3,(i+2)%3
            gx=(p[j][1].v-p[k][1].v)/twice;gy=(p[k][0].v-p[j][0].v)/twice
            for col in range(22):
                D[t,col]+=gx*m.INITIAL['R'][3*tri[i]][col]+gy*m.INITIAL['R'][3*tri[i]+1][col]
    relations=sp.Matrix.vstack(*(v.T for v in sp.Matrix(m.INITIAL['D']).T.nullspace()))
    m.require(relations.rank()==8,'eight independent constant row relations')
    for value in relations*D:
        m.require(sp.cancel(value)==0,'symbolic full divergence row identities')
    I=sp.eye(24)
    selector=sp.Matrix.vstack(*(I[row,:]for row in m.ROWS))
    m.require(selector.col_join(relations).det()!=0,'selected strong rows and relations determine all full rows')
    for col in range(22):
        value=sum(areas[t]*D[t,col]for t in range(24))-(m.INITIAL['R'][3*3+1][col]+m.INITIAL['R'][3*4+1][col])/2
        m.require(sp.cancel(value)==0,'symbolic full divergence integral and fourth cap velocity')
    return dict(symbolic_variables=['x0','x1','h0'],relations=[[int(x)for x in row]for row in relations.tolist()],
                all_176_row_identity_entries_exact_zero=True,
                selected_rows=m.ROWS.tolist(),full_row_reconstruction_exact=True,
                fourth_cap_velocity_from_integrated_divergence_exact=True)


class Interval:
    def __init__(self,lo,hi=None):
        self.lo=Q(lo);self.hi=self.lo if hi is None else Q(hi)
        m.require(self.lo<=self.hi,'ordered exact interval')
    def __add__(self,o):
        o=interval(o);return Interval(self.lo+o.lo,self.hi+o.hi)
    __radd__=__add__
    def __neg__(self):return Interval(-self.hi,-self.lo)
    def __sub__(self,o):return self+-interval(o)
    def __rsub__(self,o):return interval(o)+-self
    def __mul__(self,o):
        o=interval(o);v=[a*b for a in [self.lo,self.hi]for b in [o.lo,o.hi]];return Interval(min(v),max(v))
    __rmul__=__mul__
    def __truediv__(self,o):
        o=interval(o);m.require(o.lo>0 or o.hi<0,'interval division excludes zero');return self*Interval(1/o.hi,1/o.lo)
    def __rtruediv__(self,o):return interval(o)/self
    def __pow__(self,n):
        m.require(n in [0,1,2],'bounded interval power')
        if n==0:return Interval(1)
        if n==1:return self
        v=[self.lo*self.lo,self.hi*self.hi]
        return Interval(0 if self.lo<=0<=self.hi else min(v),max(v))
    def __lt__(self,o):return self.hi<interval(o).lo
    def __gt__(self,o):return self.lo>interval(o).hi
    def __eq__(self,o):
        o=interval(o);return self.lo==o.lo and self.hi==o.hi
    def magnitude(self):return max(abs(self.lo),abs(self.hi))


def interval(value):return value if isinstance(value,Interval)else Interval(value)


def rational_geometry(q0,eta0,alpha,time):
    q=[q0[i]+time*eta0[i]+(time*time/2)*alpha[i]for i in range(3)]
    qdot=[eta0[i]+time*alpha[i]for i in range(3)]
    cap=[(q[0],q[2]),(q[1],Q(9,4)-q[2]),(q[0]+1,q[2])]
    speed=[(qdot[0],qdot[2]),(qdot[1],-qdot[2]),(qdot[0],qdot[2])]
    points,micro,_,_,_=m.geometry(cap,speed)
    D=[[interval(0)if isinstance(time,Interval)else Q(0)for _ in range(22)]for _ in range(24)]
    min_area=[];qualities=[]
    for t,tri in enumerate(micro):
        p=[points[i]for i in tri]
        twice=m.base.cross(m.base.sub(p[1],p[0]),m.base.sub(p[2],p[0])).v
        m.require(twice>0,'whole path positive microarea')
        lengths=sum((p[i][d].v-p[(i+1)%3][d].v)**2 for i in range(3)for d in range(2))
        quality=twice/lengths
        m.require(quality>Q(1,10**6),'whole path existing minimum quality')
        min_area.append(twice/2);qualities.append(quality)
        for i in range(3):
            j,k=(i+1)%3,(i+2)%3
            gx=(p[j][1].v-p[k][1].v)/twice;gy=(p[k][0].v-p[j][0].v)/twice
            for col in range(22):D[t][col]+=gx*m.INITIAL['R'][3*tri[i]][col]+gy*m.INITIAL['R'][3*tri[i]+1][col]
    return D,min_area,qualities,points,micro


def neumann_bound(center,entries):
    inv=sp.Matrix(center).inv()
    size=len(center)
    bound=Q(0)
    for i in range(size):
        row=Q(0)
        for j in range(size):
            value=sum((Q(inv[i,k])*entries[k][j]for k in range(size)),Interval(0))-int(i==j)
            row+=value.magnitude()
        bound=max(bound,row)
    m.require(bound<1,'exact interval Neumann rank bound')
    return bound


def certify(cell,max_boxes=256):
    from cell import initial
    q,eta=initial(cell['initial_kind'])
    q0=list(map(Q,q));eta0=list(map(Q,eta));alpha=list(map(Q,cell['unknowns'][:6]));h=Q(cell['interval'])
    pending=[(Q(0),h)];boxes=[];failed=[]
    C=[[Q(x)for x in row]for row in m.C_CAP];selectors=[[Q(int(j==i))for j in range(22)]for i in m.SELECT]
    while pending:
        lo,hi=pending.pop();mid=(lo+hi)/2
        try:
            D,areas,quality,_,_=rational_geometry(q0,eta0,alpha,Interval(lo,hi))
            Dc,_,_,_,_=rational_geometry(q0,eta0,alpha,mid)
            L=[D[i]for i in m.ROWS]+[[Interval(x)for x in row]for row in C+selectors]
            Lc=[Dc[i]for i in m.ROWS]+C+selectors
            chart=neumann_bound(Lc,L)
            pressure=neumann_bound([[Dc[i][j]for j in m.PIVOT]for i in m.ROWS],[[D[i][j]for j in m.PIVOT]for i in m.ROWS])
            print('certified box',len(boxes)+1, 'width',float(hi-lo),flush=True,file=__import__('sys').stderr)
            boxes.append(dict(left=str(lo),right=str(hi),chart_neumann_bound=str(chart),pressure_neumann_bound=str(pressure),
                             chart_neumann_bound_float=float(chart),pressure_neumann_bound_float=float(pressure),
                             minimum_microarea_bound=float(min(x.lo for x in areas)),minimum_quality_bound=float(min(x.lo for x in quality))))
        except ValueError as error:
            failed.append(dict(left=str(lo),right=str(hi),reason=str(error)))
            m.require(len(boxes)+len(pending)+2<=max_boxes,'bounded interval certificate subdivisions')
            pending.extend([(mid,hi),(lo,mid)])
    boxes.sort(key=lambda box:Q(box['left']))
    m.require(Q(boxes[0]['left'])==0 and Q(boxes[-1]['right'])==h,'entire physical interval certified')
    m.require(all(a['right']==b['left']for a,b in zip(boxes,boxes[1:])),'complete interval partition')
    return dict(scope='exact rational real-path interval rank/geometry certificate, no IEEE or mesh-family theorem',
                initial_kind=cell['initial_kind'],interval=cell['interval'],boxes=boxes,rejected_coarse_boxes=failed,
                max_boxes=max_boxes,all_microareas_and_quality_positive=True,divergence_rank_16_throughout=True,
                full_material_chart_nonsingular_throughout=True,pressure_image_basis_nonsingular_throughout=True,
                row_identities=row_identities())


if __name__=='__main__':

    print(json.dumps(certify(cell),indent=2))
