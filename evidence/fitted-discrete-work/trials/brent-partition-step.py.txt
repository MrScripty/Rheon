from scipy.optimize import brentq
from functools import lru_cache

EPS=np.finfo(float).eps
MAX_EVALUATIONS=200
MAX_SIGN_ROOTS=64
MAX_ROOT_EVALUATIONS=4096

@lru_cache(maxsize=2)
def rule(order):
 m.require(order in (16,32),'bounded quadrature order')
 return np.polynomial.legendre.leggauss(order)


def initial(kind='initial'):return frozen.initial(kind)


def point(q0,eta0,unknown,time):return m.cell_point(q0,eta0,unknown[:6],time,unknown[6:])


def sign_partition(q0,eta0,unknown,h):
 # Numerical root discovery is bounded; it is not a whole-path sign theorem.
 # An unresolved/missed sign change must still fail the physical refinement gate.
 grid=np.linspace(0,h,17);values=[point(q0,eta0,unknown,t)['flux']['flux']for t in grid]
 roots=[];calls=0
 for a,b,fa,fb in zip(grid,grid[1:],values,values[1:]):
  for k in np.flatnonzero(fa*fb<0):
   def face(t):
    nonlocal calls
    calls+=1;m.require(calls<=MAX_ROOT_EVALUATIONS,'bounded face root evaluations')
    return point(q0,eta0,unknown,t)['flux']['flux'][k]
   root=brentq(face,a,b,xtol=np.nextafter(0.,1.),rtol=4*EPS,maxiter=64)
   if 0<root<h:roots.append(root)
   m.require(len(roots)<=MAX_SIGN_ROOTS,'bounded face sign roots')
 boundaries=sorted(set([0.,*roots,h]))
 return boundaries,calls


def transfers(q0,eta0,unknown,h,order,diagnostics=False,partition=None):
 plus=np.zeros(len(m.PAIR));minus=plus.copy();physical=np.zeros(22);physical_force=np.zeros(22);maxima=np.zeros(4);minimum_quality=float("inf")
 samples=[]
 boundaries,root_calls=sign_partition(q0,eta0,unknown,h)if partition is None else partition
 for left,right in zip(boundaries,boundaries[1:]):
  for node,weight in zip(*rule(order)):
   t=left+(right-left)*(1+node)/2;w=(right-left)*weight/2;p=point(q0,eta0,unknown,t);s=p['spatial'];f=p['flux']
   plus+=w*np.maximum(f['flux'],0);minus+=w*np.maximum(-f['flux'],0)
   physical+=w*np.einsum('ndj,nd->j',m.R_NODE,f['convection'])
   physical_force+=w*(np.einsum('ndj,nd->j',m.R_NODE,f['convection'])+s['K']@p['z']+s['B'].T@unknown[6:])
   edges=s['pos'][m.TRI]-np.roll(s['pos'][m.TRI],-1,axis=1)
   quality=float(np.min(2*s['area']/np.sum(edges**2,axis=(1,2))))
   minimum_quality=min(minimum_quality,quality)
   m.require(quality>=1e-6 and min(p['q'][2],2.25-p['q'][2])>=1e-10,'existing height and mesh quality gates')
   maxima=np.maximum(maxima,[np.max(np.abs(s['D']@p['z'])),np.max(np.abs(p['strong_acceleration_residual'])),np.max(np.abs(p['material_cap_residual'])),np.linalg.norm(p['momentum_residual'])])
   if diagnostics:samples.append(dict(time=float(t),q=p['q'].tolist(),divergence=(s['D']@p['z']).tolist(),acceleration_constraint=p['strong_acceleration_residual'].tolist(),material_cap=p['material_cap_residual'].tolist(),continuous_momentum=p['momentum_residual'].tolist()))
 return dict(plus=plus,minus=minus,actual_integrated_momentum_convection=physical,actual_integrated_force=physical_force,maxima=maxima,minimum_quality=minimum_quality,points=samples,partition=boundaries,root_evaluations=root_calls)


def route(plus,minus,velocity):
 routed=plus[:,None]*velocity[m.PAIR_I]-minus[:,None]*velocity[m.PAIR_J]
 convection=np.zeros((16,3));np.add.at(convection,m.PAIR_I,routed);np.add.at(convection,m.PAIR_J,-routed)
 return np.einsum('ndj,nd->j',m.R_NODE,convection)


def evaluate(q0,eta0,unknown,h,order=16,diagnostics=False,partition=None):
 a=point(q0,eta0,unknown,0);b=point(q0,eta0,unknown,h);s0=a['spatial'];s1=b['spatial'];z0=a['z'];z1=b['z']
 f=transfers(q0,eta0,unknown,h,order,diagnostics,partition);u0=a['flux']['velocity'];u1=b['flux']['velocity']
 convection=route(f['plus'],f['minus'],u1)
 for p in [a,b]:
  f['maxima']=np.maximum(f['maxima'],[np.max(np.abs(p['spatial']['D']@p['z'])),np.max(np.abs(p['strong_acceleration_residual'])),np.max(np.abs(p['material_cap_residual'])),np.linalg.norm(p['momentum_residual'])])
 res=s1['M']@z1-s0['M']@z0+convection+h*(s1['K']@z1+s1['B'].T@unknown[6:])
 gcl=s1['mass']-s0['mass'];np.add.at(gcl,m.PAIR_I,f['plus']-f['minus']);np.add.at(gcl,m.PAIR_J,-f['plus']+f['minus'])
 e0=.5*z0@s0['M']@z0;e1=.5*z1@s1['M']@z1
 dbe=.5*np.dot(s0['mass'],np.sum((u1-u0)**2,axis=1));dmix=.5*np.dot(f['plus']+f['minus'],np.sum((u1[m.PAIR_I]-u1[m.PAIR_J])**2,axis=1));dmu=h*z1@s1['K']@z1
 wp=h*z1@s1['B'].T@unknown[6:];wg=.5*np.dot(gcl,np.sum(u1**2,axis=1));wr=z1@res
 ledger=e1-e0+dbe+dmix+dmu+wp+wg-wr
 allowance=128*EPS*(e0+e1+dbe+dmix+abs(dmu)+abs(wp)+abs(wg)+abs(wr))
 difference=convection-f['actual_integrated_momentum_convection']
 return dict(start=a,end=b,transfers=f,endpoint_momentum_convection=convection,residual=res,gcl=gcl,energy_old=e0,energy_new=e1,backward_Euler_loss=dbe,mixing_loss=dmix,viscous_loss=dmu,pressure_work=wp,GCL_work=wg,discrete_residual_work=wr,energy_ledger_error=ledger,fixed_work_allowance=allowance,endpoint_vs_actual_momentum_convection=difference,actual_integrated_physical_momentum_residual=s1['M']@z1-s0['M']@z0+f['actual_integrated_force'],horizontal_momentum_change=np.dot(s1['mass'],u1[:,0])-np.dot(s0['mass'],u0[:,0]))


def seed(q0,eta0):
 s=m.spatial(m.q_to_x(q0));J,_=m.cap_chart(s);z=J@eta0
 _,p=m.evaluate(np.r_[m.q_to_x(q0),z[m.FREE]])
 return np.r_[m.C_CAP@p['acceleration'],p['acceleration'][m.SELECT],p['pressure']]


def qualify(v,h):
 m.require(np.linalg.norm(v['residual'])/h<=1e-11,'rate-normalized all 22 finite momentum equations')
 m.require(np.all(v['transfers']['maxima'][:3]<=1e-11),'all strong divergence acceleration material path gates')
 m.require(abs(v['horizontal_momentum_change'])<=1e-11,'horizontal momentum conservation')
 gallow=128*EPS*(v['start']['spatial']['mass']+v['end']['spatial']['mass'])
 m.require(np.all(np.abs(v['gcl'])<=gallow),'physical finite local GCL')
 m.require(v['backward_Euler_loss']>=0 and v['mixing_loss']>=0 and v['viscous_loss']>=0,'positive separate dissipations')
 m.require(abs(v['discrete_residual_work'])<=v['fixed_work_allowance'],'unchanged discrete momentum work gate')
 m.require(abs(v['energy_ledger_error'])<=v['fixed_work_allowance'],'complete discrete energy identity')
 m.require(abs(v['pressure_work'])<=v['fixed_work_allowance'],'endpoint pressure adjoint work')
 return gallow


def solve(h,kind='initial',state=None,diagnostics=True):
 m.require(np.isfinite(h) and 0<h<=.05,'supported research interval')
 q0,eta0=initial(kind)if state is None else state
 calls=0
 def equation(unknown):
  nonlocal calls
  calls+=1;m.require(calls<=MAX_EVALUATIONS,'bounded nonlinear evaluations')
  return evaluate(q0,eta0,unknown,h)['residual']/h
 unknown=seed(q0,eta0)
 # Stop on the actual rate residual, stricter than the acceptance gate. A
 # solver's relative-iterate stagnation flag is not a physical residual.
 converged=False
 for iteration in range(7):
  residual=equation(unknown)
  if np.linalg.norm(residual)<=1e-13:
   converged=True;break
  columns=[]
  for j in range(22):
   delta=1e-6*(abs(unknown[j])+.01)
   perturbed=unknown.copy();perturbed[j]+=delta
   columns.append((equation(perturbed)-residual)/delta)
  correction=np.linalg.solve(np.array(columns).T,-residual)
  m.require(np.all(np.isfinite(correction)),'finite Newton correction')
  unknown=unknown+correction
 m.require(converged,'bounded Newton true residual convergence')
 partition=sign_partition(q0,eta0,unknown,h)
 v16=evaluate(q0,eta0,unknown,h,partition=partition);v=evaluate(q0,eta0,unknown,h,32,diagnostics,partition)
 gallow=qualify(v,h)
 qerror=max(np.max(np.abs(v['transfers'][k]-v16['transfers'][k]))for k in ['plus','minus','actual_integrated_momentum_convection'])
 m.require(qerror<=1e-15,'physical mass/momentum quadrature refinement')
 samples=[]
 if diagnostics:
  for t in np.linspace(0,h,5):
   p=point(q0,eta0,unknown,t);s=p['spatial'];samples.append(dict(time=float(t),q=p['q'].tolist(),eta=p['eta'].tolist(),nodes=s['pos'].tolist(),velocity=p['flux']['velocity'].tolist(),reduced_velocity=p['z'].tolist(),mass=s['mass'].tolist(),pressure=(s['Q']@unknown[6:]).tolist()))
 scalar=['energy_old','energy_new','backward_Euler_loss','mixing_loss','viscous_loss','pressure_work','GCL_work','discrete_residual_work','energy_ledger_error','fixed_work_allowance','horizontal_momentum_change']
 return dict(scope='first-order endpoint donor MOMENTUM from actual path-integrated MASS; research discrete work only',initial_kind=kind,initial_q=q0.tolist(),initial_eta=eta0.tolist(),interval=h,unknowns=unknown.tolist(),nonlinear_evaluations=calls,solver_success=converged,solver_message="bounded Newton rate residual <= 1e-13",newton_iterations=iteration+1,
  **{k:float(v[k])for k in scalar},finite_momentum_residual=v['residual'].tolist(),finite_momentum_rate_norm=float(np.linalg.norm(v['residual'])/h),minimum_sampled_quality=v['transfers']['minimum_quality'],full_path_maxima=v['transfers']['maxima'].tolist(),actual_integrated_physical_momentum_residual=v['actual_integrated_physical_momentum_residual'].tolist(),actual_integrated_physical_momentum_norm=float(np.linalg.norm(v['actual_integrated_physical_momentum_residual'])),actual_physical_integrated_momentum_gate_pass=bool(np.linalg.norm(v['actual_integrated_physical_momentum_residual'])<=1e-11),continuous_momentum_gate_pass=bool(v['transfers']['maxima'][3]<=1e-11),
  actual_integrated_physical_momentum_convection=v['transfers']['actual_integrated_momentum_convection'].tolist(),endpoint_donor_momentum_convection=v['endpoint_momentum_convection'].tolist(),endpoint_vs_actual_momentum_max=float(np.max(np.abs(v['endpoint_vs_actual_momentum_convection']))),
  actual_local_GCL_defect=v['gcl'].tolist(),local_GCL_allowance=gallow.tolist(),quadrature_16_32_difference=float(qerror),positive_negative_face_integrals=[[i,j,float(v['transfers']['plus'][k]),float(v['transfers']['minus'][k])]for k,(i,j)in enumerate(m.PAIR)],
  end_q=v['end']['q'].tolist(),end_eta=v['end']['eta'].tolist(),end_reduced_velocity=v['end']['z'].tolist(),end_pressure=(v['end']['spatial']['Q']@unknown[6:]).tolist(),discrete_work_gate_pass=True,research_discrete_candidate_qualified=True,new_public_step_enabled=False,face_sign_partition=v['transfers']['partition'],face_root_evaluations=v['transfers']['root_evaluations'],quadrature_points=v['transfers']['points'],samples=samples)

if __name__=='__main__':
